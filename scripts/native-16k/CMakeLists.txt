cmake_minimum_required(VERSION 3.22.1)
project(flipper_sample_native_16k C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384")
if(NOT SOURCE_ROOT)
  message(FATAL_ERROR "Set SOURCE_ROOT to the directory prepared by rebuild-sample-native.ps1")
endif()
file(TO_CMAKE_PATH "${SOURCE_ROOT}" SOURCE_ROOT)

add_subdirectory("${SOURCE_ROOT}/yoga-2.0.1/java" yoga)
set(FLEX "${SOURCE_ROOT}/litho-0.50.1/litho-flexlayout/src/main")
file(GLOB_RECURSE FLEX_CORE "${FLEX}/cpp/flexlayout/*.cpp")
file(GLOB FLEX_JNI "${FLEX}/java/jni/*.cpp")
add_library(flexlayout SHARED ${FLEX_CORE} ${FLEX_JNI})
set_property(TARGET flexlayout PROPERTY CXX_STANDARD 20)
target_include_directories(flexlayout PRIVATE "${FLEX}/cpp" "${FLEX}/cpp/flexlayout")
target_link_libraries(flexlayout android log)

set(FRESCO "${SOURCE_ROOT}/fresco-3.1.3")
set(IMAGE "${FRESCO}/imagepipeline-native/src/main/jni")
file(GLOB IMAGE_SOURCES "${IMAGE}/imagepipeline/*.cpp" "${IMAGE}/bitmaps/*.c" "${IMAGE}/memchunk/*.c")
add_library(imagepipeline SHARED ${IMAGE_SOURCES})
target_include_directories(imagepipeline PRIVATE "${IMAGE}/imagepipeline" "${IMAGE}/bitmaps" "${IMAGE}/memchunk")
target_compile_definitions(imagepipeline PRIVATE LOG_TAG="libimagepipeline")
target_compile_options(imagepipeline PRIVATE -fvisibility=hidden)
target_link_libraries(imagepipeline log jnigraphics)

set(FILTERS "${FRESCO}/native-filters/src/main/jni")
file(GLOB FILTER_SOURCES "${FILTERS}/native-filters/*.cpp" "${FILTERS}/filters/*.c")
add_library(native-filters SHARED ${FILTER_SOURCES})
target_include_directories(native-filters PRIVATE "${FILTERS}/native-filters" "${FILTERS}/filters")
target_compile_definitions(native-filters PRIVATE LOG_TAG="libnative-filters")
target_compile_options(native-filters PRIVATE -fvisibility=hidden)
target_link_libraries(native-filters log jnigraphics)

set(JPEG "${SOURCE_ROOT}/libjpeg-turbo-1.5.3")
set(JPEG_CONFIG "${FRESCO}/native-imagetranscoder/src/main/jni/third-party/libjpeg-turbo-1.5.3")
set(JPEG_FILES jcapimin.c jcapistd.c jccoefct.c jccolor.c jcdctmgr.c jchuff.c jcinit.c jcmainct.c jcmarker.c jcmaster.c
    jcomapi.c jcparam.c jcphuff.c jcprepct.c jcsample.c jctrans.c jdapimin.c jdapistd.c jdatadst.c jdatasrc.c
    jdcoefct.c jdcolor.c jddctmgr.c jdhuff.c jdinput.c jdmainct.c jdmarker.c jdmaster.c jdmerge.c jdphuff.c
    jdpostct.c jdsample.c jdtrans.c jerror.c jfdctflt.c jfdctfst.c jfdctint.c jidctflt.c jidctfst.c jidctint.c
    jidctred.c jquant1.c jquant2.c jutils.c jmemmgr.c jaricom.c jcarith.c jdarith.c transupp.c jmemnobs.c jsimd_none.c)
list(TRANSFORM JPEG_FILES PREPEND "${JPEG}/")
add_library(jpeg-static STATIC ${JPEG_FILES})
target_include_directories(jpeg-static PUBLIC "${JPEG_CONFIG}" "${JPEG}")
set(TRANSCODER "${FRESCO}/native-imagetranscoder/src/main/jni/native-imagetranscoder")
file(GLOB TRANSCODER_SOURCES "${TRANSCODER}/*.cpp" "${TRANSCODER}/jpeg/*.cpp")
add_library(native-imagetranscoder SHARED ${TRANSCODER_SOURCES})
target_include_directories(native-imagetranscoder PRIVATE "${TRANSCODER}")
target_compile_definitions(native-imagetranscoder PRIVATE LOG_TAG="libnative-imagetranscoder")
target_compile_options(native-imagetranscoder PRIVATE -fvisibility=hidden)
target_link_libraries(native-imagetranscoder jpeg-static log)

set(ART "${SOURCE_ROOT}/androidx/inspection/inspection/src/main/native")
file(GLOB SLICER "${SOURCE_ROOT}/dexter/slicer/*.cc")
file(GLOB ART_SOURCES "${ART}/inspection/*.cc" "${ART}/jvmti/*.cc" "${ART}/util/*.cc")
configure_file("$ENV{JAVA_HOME}/include/jvmti.h" "${CMAKE_CURRENT_BINARY_DIR}/jvmti/jvmti.h" COPYONLY)
add_library(art_tooling SHARED ${ART_SOURCES} ${SLICER})
target_include_directories(art_tooling PRIVATE "${ART}/jvmti" "${ART}/util" "${SOURCE_ROOT}/dexter/slicer/export" "${CMAKE_CURRENT_BINARY_DIR}/jvmti")
target_link_libraries(art_tooling log z)
